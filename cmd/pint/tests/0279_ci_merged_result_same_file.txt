# Setup:
# 1. main has rules.yml with rule1, rule2 and rule3.
# 2. The MR branch starts from main and breaks rule1.
# 3. Later, main gets a new commit that breaks rule3. The MR does not
#    include this commit.
# 4. GitLab merges the MR into the latest main and runs CI on the result.
#    So the checkout has both broken rules.
#
# Expected: pint reports only rule1. The MR did not change rule3.

mkdir testrepo
cd testrepo
exec git init --initial-branch=main .

cp ../src/.pint.hcl .
cp ../src/rules.yml rules.yml
env GIT_AUTHOR_NAME=pint
env GIT_AUTHOR_EMAIL=pint@example.com
env GIT_COMMITTER_NAME=pint
env GIT_COMMITTER_EMAIL=pint@example.com
exec git add .
exec git commit -am 'init'

exec git checkout -b feature
cp ../src/feature.yml rules.yml
exec git commit -am 'feature change'

exec git checkout main
cp ../src/main.yml rules.yml
exec git commit -am 'main change'

exec git checkout --detach
exec git merge --no-ff feature -m 'merged result'

! exec pint --no-color ci
stderr 'msg="Problems found" Fatal=1$'
stderr '---> rules\.yml:5 -> `rule1`'
! stderr 'rule3'

-- src/.pint.hcl --
ci {
  baseBranch = "main"
}
-- src/rules.yml --
groups:
- name: rules
  rules:
  - record: rule1
    expr: sum(foo) by(job)
  - record: rule2
    expr: sum(foo) by(job)
  - record: rule3
    expr: sum(foo) by(job)
-- src/feature.yml --
groups:
- name: rules
  rules:
  - record: rule1
    expr: sum(foo) bi()
  - record: rule2
    expr: sum(foo) by(job)
  - record: rule3
    expr: sum(foo) by(job)
-- src/main.yml --
groups:
- name: rules
  rules:
  - record: rule1
    expr: sum(foo) by(job)
  - record: rule2
    expr: sum(foo) by(job)
  - record: rule3
    expr: sum(foo) bi()
