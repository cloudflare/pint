# Setup:
# 1. main has rules.yml with rule1, rule2 and rule3.
# 2. A sub branch starts from main and breaks rule3.
# 3. Later, main gets a new commit that breaks rule2.
# 4. The MR branch starts from the latest main.
# 5. The MR author merges the sub branch into the MR branch, then breaks
#    rule1.
# 6. CI runs on the MR branch.
#
# Expected: pint reports rule1 and rule3. rule2 was changed on main, not
# in the MR.

mkdir testrepo
cd testrepo
exec git init --initial-branch=main .

cp ../src/.pint.hcl .
cp ../src/rules.yml rules.yml
env GIT_AUTHOR_NAME=pint
env GIT_AUTHOR_EMAIL=pint@example.com
env GIT_COMMITTER_NAME=pint
env GIT_COMMITTER_EMAIL=pint@example.com
exec git add .
exec git commit -am 'init'

exec git checkout -b sub
cp ../src/sub.yml rules.yml
exec git commit -am 'sub change'

exec git checkout main
cp ../src/main.yml rules.yml
exec git commit -am 'main change'

exec git checkout -b feature
exec git merge --no-ff sub -m 'merge sub into feature'
cp ../src/feature.yml rules.yml
exec git commit -am 'feature change'

! exec pint --no-color ci --show-duplicates
stderr 'msg="Problems found" Fatal=2$'
stderr '---> rules\.yml:5 -> `rule1`'
stderr '---> rules\.yml:9 -> `rule3`'
! stderr 'rule2'

-- src/.pint.hcl --
ci {
  baseBranch = "main"
}
-- src/rules.yml --
groups:
- name: rules
  rules:
  - record: rule1
    expr: sum(foo) by(job)
  - record: rule2
    expr: sum(bar) by(job)
  - record: rule3
    expr: sum(baz) by(job)
-- src/sub.yml --
groups:
- name: rules
  rules:
  - record: rule1
    expr: sum(foo) by(job)
  - record: rule2
    expr: sum(bar) by(job)
  - record: rule3
    expr: sum(baz) bi()
-- src/main.yml --
groups:
- name: rules
  rules:
  - record: rule1
    expr: sum(foo) by(job)
  - record: rule2
    expr: sum(bar) bi()
  - record: rule3
    expr: sum(baz) by(job)
-- src/feature.yml --
groups:
- name: rules
  rules:
  - record: rule1
    expr: sum(foo) bi()
  - record: rule2
    expr: sum(bar) bi()
  - record: rule3
    expr: sum(baz) bi()
