# Setup:
# 1. main has rules.yml.
# 2. The MR branch starts from main and breaks rule1 in rules.yml.
# 3. Later, main gets a new commit that adds ignored1.yml, ignored2.yml
#    and ignored3.yml. The MR does not include this commit.
# 4. GitLab merges the MR into the latest main and runs CI on the result.
#    So the checkout has the MR change and the new files from main.
#
# Expected: pint checks only rules.yml and reports rule1. The MR did not
# add the ignored files, so pint must not check them.

mkdir testrepo
cd testrepo
exec git init --initial-branch=main .

cp ../src/.pint.hcl .
cp ../src/rules.yml rules.yml
env GIT_AUTHOR_NAME=pint
env GIT_AUTHOR_EMAIL=pint@example.com
env GIT_COMMITTER_NAME=pint
env GIT_COMMITTER_EMAIL=pint@example.com
exec git add .
exec git commit -am 'init'

exec git checkout -b feature
cp ../src/feature.yml rules.yml
exec git add .
exec git commit -am 'feature change'

exec git checkout main
cp ../src/ignored1.yml ignored1.yml
cp ../src/ignored2.yml ignored2.yml
cp ../src/ignored3.yml ignored3.yml
exec git add .
exec git commit -am 'main change'

exec git checkout --detach
exec git merge --no-ff feature -m 'merged result'

! exec pint --no-color ci
! stdout .
! stderr 'ignored[123]'
cmp stderr ../stderr.txt
