# Setup:
# 1. main has no rules files.
# 2. The MR branch adds rüles.yml with a broken rule1. The file name has
#    a non-ASCII letter.
#
# Expected: pint checks rüles.yml and reports rule1.

mkdir testrepo
cd testrepo
exec git init --initial-branch=main .

cp ../src/.pint.hcl .
env GIT_AUTHOR_NAME=pint
env GIT_AUTHOR_EMAIL=pint@example.com
env GIT_COMMITTER_NAME=pint
env GIT_COMMITTER_EMAIL=pint@example.com
exec git add .
exec git commit -am 'init'

exec git checkout -b v2
cp ../src/rüles.yml rüles.yml
exec git add .
exec git commit -am 'add rules'

! exec pint --no-color ci
stderr 'msg="Git file changes parsed" files=1 entries=1 '
stderr '---> rüles\.yml:5 -> `rule1`'

-- src/.pint.hcl --
ci {
  baseBranch = "main"
}
-- src/rüles.yml --
groups:
- name: rules
  rules:
  - record: rule1
    expr: sum(foo) bi()
