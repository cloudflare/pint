# Setup:
# 1. main has rules.yml with rule1.
# 2. The MR branch breaks rule1.
# 3. The repository has color.ui=always in its git config.
#
# Expected: pint finds that line 5 of rules.yml was modified. The git
# config must not change what pint reads from git.

mkdir testrepo
cd testrepo
exec git init --initial-branch=main .

cp ../src/.pint.hcl .
cp ../src/rules.yml rules.yml
env GIT_AUTHOR_NAME=pint
env GIT_AUTHOR_EMAIL=pint@example.com
env GIT_COMMITTER_NAME=pint
env GIT_COMMITTER_EMAIL=pint@example.com
exec git add .
exec git commit -am 'init'

exec git checkout -b v2
cp ../src/feature.yml rules.yml
exec git commit -am 'feature change'

exec git config color.ui always
! exec pint --no-color -l debug ci
stderr 'msg="File was modified" path=rules\.yml lines=\[\{"Before":5,"After":5,"Modified":true\}\]'

ci {
  baseBranch = "main"
}
-- src/rules.yml --

- name: rules
  rules:
  - record: rule1
    expr: sum(foo) by(job)
