# Setup:
# 1. main has rules.yml.
# 2. Branch dep starts from main and breaks rule1.
# 3. The MR branch starts from main, merges dep, then adds extra.yml.
#    So the rule1 change from dep is part of the MR.
#
# Expected: pint reports rule1.

mkdir testrepo
cd testrepo
exec git init --initial-branch=main .

cp ../src/.pint.hcl .
cp ../src/rules.yml rules.yml
env GIT_AUTHOR_NAME=pint
env GIT_AUTHOR_EMAIL=pint@example.com
env GIT_COMMITTER_NAME=pint
env GIT_COMMITTER_EMAIL=pint@example.com
exec git add .
exec git commit -am 'init'

exec git checkout -b dep
cp ../src/dep.yml rules.yml
exec git commit -am 'dep change'

exec git checkout main
exec git checkout -b feature
exec git merge --no-ff dep -m 'merge dep into feature'
cp ../src/extra.yml extra.yml
exec git add .
exec git commit -am 'feature change'

! exec pint --no-color ci
stderr '---> rules\.yml:5 -> `rule1`'

-- src/.pint.hcl --
ci {
  baseBranch = "main"
}
-- src/rules.yml --
groups:
- name: rules
  rules:
  - record: rule1
    expr: sum(foo) by(job)
-- src/dep.yml --
groups:
- name: rules
  rules:
  - record: rule1
    expr: sum(foo) bi()
-- src/extra.yml --
groups:
- name: extra
  rules:
  - record: extra
    expr: sum(foo) by(job)
