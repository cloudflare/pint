# Setup:
# 1. main has rules.yml.
# 2. The MR branch starts from main and breaks rule1.
# 3. Later, main gets a new commit that adds ignored.yml.
# 4. GitLab merges the MR into the latest main and runs CI on the result.
# 5. CI uses a shallow clone that does not have the commit where the MR
#    started.
#
# Expected:
# - pint cannot find where the MR started. It exits with an error that
#   tells the user to fetch more git history.
# - After more history is fetched, pint checks only rules.yml and reports
#   rule1. The MR did not change .pint.hcl or ignored.yml, so pint must not
#   check them.

mkdir testrepo
cd testrepo
exec git init --initial-branch=main .

cp ../src/.pint.hcl .
cp ../src/rules.yml rules.yml
env GIT_AUTHOR_NAME=pint
env GIT_AUTHOR_EMAIL=pint@example.com
env GIT_COMMITTER_NAME=pint
env GIT_COMMITTER_EMAIL=pint@example.com
exec git add .
exec git commit -am 'init'

exec git checkout -b feature
cp ../src/feature.yml rules.yml
exec git commit -am 'feature change'

exec git checkout main
cp ../src/ignored.yml ignored.yml
exec git add .
exec git commit -am 'main change'

exec git checkout --detach
exec git merge --no-ff feature -m 'merged result'
exec git branch merged

cd ..
exec git clone --quiet --depth 2 --branch merged file://$WORK/testrepo clone
cd clone
exec git fetch --quiet --depth 1 origin main:main
exec git checkout --quiet --detach

! exec pint --no-color ci
stderr 'failed to find the common ancestor of main and .+, make sure both are fetched with enough git history'

exec git fetch --quiet --deepen 1 origin
! exec pint --no-color ci
stderr 'msg="Git file changes parsed" files=1 entries=1 '
stderr 'msg="Problems found" Fatal=1$'
stderr '---> rules\.yml:5 -> `rule1`'
! stderr 'ignored'

-- src/.pint.hcl --
ci {
  baseBranch = "main"
}
-- src/rules.yml --
groups:
- name: rules
  rules:
  - record: rule1
    expr: sum(foo) by(job)
-- src/feature.yml --
groups:
- name: rules
  rules:
  - record: rule1
    expr: sum(foo) bi()
-- src/ignored.yml --
groups:
- name: rules
  rules:
  - record: ignored
    expr: sum(foo) bi()
