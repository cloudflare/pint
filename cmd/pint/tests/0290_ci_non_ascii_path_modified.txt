# Setup:
# 1. main has rüles.yml with rule1. The file name has a non-ASCII letter.
# 2. The MR branch breaks rule1 in rüles.yml.
#
# Expected: pint finds that line 5 of rüles.yml was modified and reports
# rule1.

mkdir testrepo
cd testrepo
exec git init --initial-branch=main .

cp ../src/.pint.hcl .
cp ../src/rules.yml rüles.yml
env GIT_AUTHOR_NAME=pint
env GIT_AUTHOR_EMAIL=pint@example.com
env GIT_COMMITTER_NAME=pint
env GIT_COMMITTER_EMAIL=pint@example.com
exec git add .
exec git commit -am 'init'

exec git checkout -b v2
cp ../src/feature.yml rüles.yml
exec git commit -am 'feature change'

! exec pint --no-color -l debug ci
stderr 'msg="File was modified" path=rüles\.yml lines=\[\{"Before":5,"After":5,"Modified":true\}\]'
stderr '---> rüles\.yml:5 -> `rule1`'

-- src/.pint.hcl --
ci {
  baseBranch = "main"
}
-- src/rules.yml --
groups:
- name: rules
  rules:
  - record: rule1
    expr: sum(foo) by(job)
-- src/feature.yml --
groups:
- name: rules
  rules:
  - record: rule1
    expr: sum(foo) bi()
