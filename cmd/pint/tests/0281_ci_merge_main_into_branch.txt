# Setup:
# 1. main has rules.yml.
# 2. The MR branch starts from main and breaks rule1.
# 3. Later, main gets a new commit that adds ignored.yml.
# 4. The MR author merges main into the MR branch to get that commit.
# 5. CI runs on the MR branch.
#
# Expected: pint checks only rules.yml and reports rule1. ignored.yml came
# from main, so pint must not check it.

mkdir testrepo
cd testrepo
exec git init --initial-branch=main .

cp ../src/.pint.hcl .
cp ../src/rules.yml rules.yml
env GIT_AUTHOR_NAME=pint
env GIT_AUTHOR_EMAIL=pint@example.com
env GIT_COMMITTER_NAME=pint
env GIT_COMMITTER_EMAIL=pint@example.com
exec git add .
exec git commit -am 'init'

exec git checkout -b feature
cp ../src/feature.yml rules.yml
exec git commit -am 'feature change'

exec git checkout main
cp ../src/ignored.yml ignored.yml
exec git add .
exec git commit -am 'main change'

exec git checkout feature
exec git merge --no-ff main -m 'merge main into feature'

! exec pint --no-color ci
stderr 'msg="Git file changes parsed" files=1 entries=1 '
stderr 'msg="Problems found" Fatal=1$'
stderr '---> rules\.yml:5 -> `rule1`'
! stderr 'ignored'

-- src/.pint.hcl --
ci {
  baseBranch = "main"
}
-- src/rules.yml --
groups:
- name: rules
  rules:
  - record: rule1
    expr: sum(foo) by(job)
-- src/feature.yml --
groups:
- name: rules
  rules:
  - record: rule1
    expr: sum(foo) bi()
-- src/ignored.yml --
groups:
- name: rules
  rules:
  - record: ignored
    expr: sum(foo) bi()
