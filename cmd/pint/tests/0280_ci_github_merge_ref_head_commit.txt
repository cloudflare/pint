# Setup:
# 1. main has no rules files.
# 2. The PR branch starts from main and adds rules.yml with a broken rule1.
# 3. Later, main gets a new commit that adds other.yml.
# 4. GitHub merges the PR into the latest main and runs CI on the result.
#    This merge commit is not part of the PR.
#
# Expected: pint posts its GitHub review and its comment for rule1 on the
# last commit of the PR branch, not on the merge commit.

http method github GET /api/v3/user 200 {"id":42,"login":"ci-user"}
http method github GET /api/v3/repos/foo/bar/pulls/123/reviews 200 []
http method github POST /api/v3/repos/foo/bar/pulls/123/reviews 200 {}
http method github GET /api/v3/repos/foo/bar/pulls/123/comments 200 []
http method github GET /api/v3/repos/foo/bar/issues/123/comments 200 []
http method github POST /api/v3/repos/foo/bar/pulls/123/comments 200 {}
http start github 127.0.0.1:6280

mkdir testrepo
cd testrepo
exec git init --initial-branch=main .

cp ../src/.pint.hcl .
env GIT_AUTHOR_NAME=pint
env GIT_AUTHOR_EMAIL=pint@example.com
env GIT_COMMITTER_NAME=pint
env GIT_COMMITTER_EMAIL=pint@example.com
exec git add .
exec git commit -am 'init'

exec git checkout -b feature
cp ../src/rules.yml rules.yml
exec git add .
exec git commit -am 'feature change'

exec git checkout main
cp ../src/other.yml other.yml
exec git add .
exec git commit -am 'main change'

exec git checkout --detach
exec git merge --no-ff feature -m 'merge feature into main'

env GITHUB_AUTH_TOKEN=12345
env GITHUB_ACTION=YES
env GITHUB_EVENT_NAME=pull_request
env GITHUB_REF=refs/pull/123/merge
env GITHUB_BASE_REF=main
env GITHUB_HEAD_REF=feature
env GITHUB_REPOSITORY=foo/bar
env GITHUB_API_URL=http://127.0.0.1:6280
! exec pint --offline --no-color -l debug ci
grep 'POST /api/v3/repos/foo/bar/pulls/123/reviews' github.got
grep 'POST /api/v3/repos/foo/bar/pulls/123/comments' github.got
cp stderr ../pint.log
exec sh -c 'grep "msg=\"Creating a review\" commit=$(git rev-parse HEAD^2) " ../pint.log'
exec sh -c 'grep "msg=\"Creating a pr comment\" commit=$(git rev-parse HEAD^2) path=rules.yml " ../pint.log'

  rules:
  - alert: rule1
    expr: sum(foo) bi()

- name: other
  rules:
  - record: other
    expr: sum(foo) by(job)
